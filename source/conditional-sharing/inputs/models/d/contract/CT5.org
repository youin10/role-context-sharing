prefix logistics: <http://ai.uos.ac.kr/logistics-ontology#>;

role BipedMonitor {
  reads {
    logistics:RobotStatus($robot, $battery, $zone) {
      $robot == logistics:Biped && $battery <= 30;
    }
  }
}

role FleetMonitor {
  reads {
    logistics:RobotStatus($robot, $battery, $zone) {
      $robot == logistics:Robot && $battery <= 40;
    }
  }
}

role Robot {
  changes {
    logistics:RobotStatus($robot, $battery, $zone);
  }
}

role Observer {
  changes {
    logistics:RobotStopped($robot, $elapsed);
  }
}
