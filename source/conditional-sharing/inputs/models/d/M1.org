prefix logistics: <http://ai.uos.ac.kr/logistics-ontology#>;

role Manager {
  reads {
    logistics:RobotStatus($robot, $battery, $zone) {
      $robot == logistics:Biped && $battery <= 30 || $robot == logistics:Wheeled && $battery <= 20;
    }
    logistics:RobotStopped($robot, $elapsed) {
      $elapsed >= 300;
    }
  }
}

role Robot {
  changes {
    logistics:RobotStatus($robot, $battery, $zone);
    logistics:CargoCarriedBy($cargo, $robot);
  }
}

role Observer {
  changes {
    logistics:RobotStopped($robot, $elapsed);
  }
}

role SafetyOfficer {
  reads {
    logistics:RobotStatus($robot, $battery, $zone) {
      $battery <= 40;
    }
  }
}

role Dispatcher {
  reads {
    logistics:CargoCarriedBy($cargo, $robot);
  }
}

role BipedTech {
  reads {
    logistics:RobotStatus($robot, $battery, $zone) {
      $robot == "biped1";
    }
  }
}

role CarryAuditor {
  reads {
    logistics:CargoCarriedBy($cargo, $robot) {
      $robot == logistics:Biped;
    }
  }
}

role NonBipedWatch {
  reads {
    logistics:RobotStatus($robot, $battery, $zone) {
      $robot != logistics:Biped;
    }
  }
}
