prefix logistics: <http://ai.uos.ac.kr/logistics-ontology#>;

role TransitCoordinator {
  reads {
    logistics:RobotLocatedAt($robot, $node) {
      $node == "t_w" || $node == "t_e" || $robot == logistics:Wheeled && $node == "apron";
    }
  }
}

role Carrier {
  changes {
    logistics:RobotLocatedAt($robot, $node);
  }
}
